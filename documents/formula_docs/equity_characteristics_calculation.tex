\begin{abstract}
This document records the working formula definitions for U.S. equity
characteristics in the EquityChars project. The table keeps the formulas that
match the current CIZ implementation and includes portfolio signal directions used in long-short portfolio construction.
\end{abstract}

\section{Notation}

Firm is indexed by \(i\), calendar month or fiscal period by \(t\), and daily
observations by \(d\). Unless stated otherwise, \(\Delta X_{i,t}=X_{i,t}-
X_{i,t-1}\). \(\mathrm{ME}\) denotes market equity, \(\mathrm{BE}\) denotes book
equity, and industry-adjusted variables subtract the cross-sectional industry
mean at time \(t\).

\(\mathrm{abr}\) &
\(\displaystyle \mathrm{ABR}_{i,t}=\sum_{d=-2}^{1}\left(\mathrm{RET}_{i,t+d}-\mathrm{RET}^{\mathrm{SP500}}_{t+d}\right)\)  & \(+1\)\\

\(\mathrm{acc}\) &
\(\displaystyle \mathrm{ACC}_{i,t}=
\begin{cases}
\dfrac{(\Delta\mathrm{ACT}_{i,t}-\Delta\mathrm{CHE}_{i,t})
-\Delta\mathrm{LCT}_{i,t}+\Delta\mathrm{DLC}_{i,t}+\Delta\mathrm{TXP}_{i,t}
-\mathrm{DP}_{i,t}}
{\frac{1}{2}(\mathrm{AT}_{i,t}+\mathrm{AT}_{i,t-1})}, & \mathrm{OANCF}_{i,t}\text{ missing},\\[0.6em]
\dfrac{\mathrm{NI}_{i,t}-\mathrm{OANCF}_{i,t}}
{\frac{1}{2}(\mathrm{AT}_{i,t}+\mathrm{AT}_{i,t-1})}, & \text{otherwise}.
\end{cases}\)  & \(-1\)\\

\(\mathrm{absacc}\) &
\(\displaystyle \mathrm{ABSACC}_{i,t}=|\mathrm{ACC}_{i,t}|\)  & \(-1\)\\

\(\mathrm{adm}\) &
\(\displaystyle \mathrm{ADM}_{i,t}=\frac{\mathrm{XAD}_{i,t}}{\mathrm{ME}_{i,t}}\).  & \(+1\)\\

\(\mathrm{age}\) &
\(\displaystyle \mathrm{AGE}_{i,t}=\mathrm{count}_{i,t}\), the Compustat history counter used in the CIZ pipeline. & \(-1\)\\

\(\mathrm{agr}\) &
\(\displaystyle \mathrm{AGR}_{i,t}=\frac{\Delta\mathrm{AT}_{i,t}}{\mathrm{AT}_{i,t-1}}\)  & \(-1\)\\

\(\mathrm{alm}\) &
\(\displaystyle
\begin{aligned}
\mathrm{ALM}_{i,t}&=\frac{\mathrm{ALA}_{i,t}}
{\mathrm{AT}_{i,t}+\mathrm{PRCC\_F}_{i,t}\mathrm{CSHO}_{i,t}-\mathrm{CEQ}_{i,t}},\\
\mathrm{ALA}_{i,t}&=\mathrm{CHE}_{i,t}
+0.75(\mathrm{ACT}_{i,t}-\mathrm{CHE}_{i,t})\\
&\quad-0.5(\mathrm{AT}_{i,t}-\mathrm{ACT}_{i,t}-\mathrm{GDWL}_{i,t}-\mathrm{INTAN}_{i,t}).
\end{aligned}\)  & \(-1\)\\

\(\mathrm{ato}\) &
\(\displaystyle \mathrm{ATO}^{A}_{i,t}=\frac{\mathrm{SALE}_{i,t}}{\mathrm{NOA}^{raw}_{i,t-1}},\quad
\mathrm{ATO}^{Q}_{i,t}=\frac{\mathrm{SALEQ}_{i,t}}{\mathrm{NOA}^{raw}_{i,t-4}}\)  & \(+1\)\\

\(\mathrm{baspread}\) &
\(\displaystyle \mathrm{BASpread}_{i,t}=
\frac{1}{|W_{i,t}|}\sum_{s\in W_{i,t}}
\frac{\mathrm{AskHi}_{i,s}-\mathrm{BidLo}_{i,s}}
{(\mathrm{AskHi}_{i,s}+\mathrm{BidLo}_{i,s})/2}\)  & \(+1\)\\

\(\mathrm{beta}\) &
\(\displaystyle \mathrm{BETA}_{i,t}=
\frac{\sum_{s\in W_{i,t}}(\mathrm{MKTRF}_s-\overline{\mathrm{MKTRF}})
(\mathrm{EXRET}_{i,s}-\overline{\mathrm{EXRET}}_i)}
{\sum_{s\in W_{i,t}}(\mathrm{MKTRF}_s-\overline{\mathrm{MKTRF}})^2}\)  & \(+1\)\\

\(\mathrm{bm}\) &
\(\displaystyle \mathrm{ME}_{i,t}=\mathrm{PRC}_{i,t}\mathrm{SHROUT}_{i,t},\quad
\mathrm{BE}_{i,t}=\mathrm{SEQ}_{i,t}+\mathrm{TXDITC}_{i,t}-\mathrm{PSTK}_{i,t},\quad
\mathrm{BM}_{i,t}=\frac{\mathrm{BE}_{i,t}}{\mathrm{ME}_{i,t}}\).  & \(+1\)\\

\(\mathrm{bm\_ia}\) &
\(\displaystyle \mathrm{BM\_IA}_{i,t}=\mathrm{BM}_{i,t}
-\frac{1}{N_{\mathrm{ind},t}}\sum_{j\in\mathrm{ind}(i)}\mathrm{BM}_{j,t}\)  & \(+1\)\\

\(\mathrm{cash}\) &
\(\displaystyle \mathrm{CASH}_{i,t}=\frac{\mathrm{CHE}_{i,t}}{\mathrm{AT}_{i,t}}\)  & \(+1\)\\

\(\mathrm{cashdebt}\) &
\(\displaystyle \mathrm{CASHDEBT}_{i,t}=
\frac{\mathrm{IB}_{i,t}+\mathrm{DP}_{i,t}}
{\frac{1}{2}(\mathrm{LT}_{i,t}+\mathrm{LT}_{i,t-1})}\)  & \(+1\)\\

\(\mathrm{cashpr}\) &
\(\displaystyle \mathrm{CASHPR}_{i,t}=
\frac{\mathrm{ME}_{i,t}+\mathrm{DLTT}_{i,t}-\mathrm{AT}_{i,t}}{\mathrm{CHE}_{i,t}}\)  & \(-1\)\\

\(\mathrm{cfp}\) &
\(\displaystyle
\mathrm{CFP}_{i,t}=\begin{cases}
\mathrm{IB}_{i,t}/\mathrm{ME}_{i,t}, & \mathrm{DP}_{i,t}\text{ missing},\\
\mathrm{NaN}, & \mathrm{IB}_{i,t}\text{ missing},\\
(\mathrm{IB}_{i,t}+\mathrm{DP}_{i,t})/\mathrm{ME}_{i,t}, & \text{otherwise}.
\end{cases}\)  & \(+1\)\\

\(\mathrm{cfp\_ia}\) &
\(\displaystyle \mathrm{CFP\_IA}_{i,t}=\mathrm{CFP}_{i,t}-\overline{\mathrm{CFP}}_{\mathrm{industry},t}\).  & \(+1\)\\

\(\mathrm{chatoia}\) &
\(\displaystyle
\begin{aligned}
\mathrm{NOA}^{raw}_{i,t}&=(\mathrm{AT}_{i,t}-\mathrm{CHE}_{i,t}-\mathrm{IVAO}_{i,t})\\
&\quad-(\mathrm{AT}_{i,t}-\mathrm{DLC}_{i,t}-\mathrm{DLTT}_{i,t}-\mathrm{MIB}_{i,t}-\mathrm{PSTK}_{i,t}-\mathrm{CEQ}_{i,t}),\\
\mathrm{CHATO}_{i,t}&=
\frac{\mathrm{SALE}_{i,t}}{\frac{1}{2}(\mathrm{NOA}^{raw}_{i,t}+\mathrm{NOA}^{raw}_{i,t-1})}
-\frac{\mathrm{SALE}_{i,t-1}}{\frac{1}{2}(\mathrm{NOA}^{raw}_{i,t-1}+\mathrm{NOA}^{raw}_{i,t-2})},\\
\mathrm{CHATOIA}_{i,t}&=\mathrm{CHATO}_{i,t}-\overline{\mathrm{CHATO}}_{\mathrm{industry},t}.
\end{aligned}\)  & \(+1\)\\

\(\mathrm{chcsho}\) &
\(\displaystyle \mathrm{CHCSHO}_{i,t}=\frac{\mathrm{CSHO}_{i,t}}{\mathrm{CSHO}_{i,t-1}}-1\)  & \(-1\)\\

\(\mathrm{chempia}\) &
\(\displaystyle
\mathrm{HIRE}_{i,t}=\begin{cases}
(\mathrm{EMP}_{i,t}-\mathrm{EMP}_{i,t-1})/\mathrm{EMP}_{i,t-1}, & \text{available},\\
0, & \text{otherwise},
\end{cases}\quad
\mathrm{CHEMPIA}_{i,t}=\mathrm{HIRE}_{i,t}-\overline{\mathrm{HIRE}}_{\mathrm{industry},t}\)  & \(-1\)\\

\(\mathrm{chpm}\) &
\(\displaystyle \mathrm{CHPM}_{i,t}=\frac{\mathrm{IB}_{i,t}}{\mathrm{SALE}_{i,t}}
-\frac{\mathrm{IB}_{i,t-1}}{\mathrm{SALE}_{i,t-1}}\)  & \(+1\)\\

\(\mathrm{chtx}\) &
\(\displaystyle \mathrm{CHTX}_{i,t}=\frac{\mathrm{TXT}_{i,t}-\mathrm{TXT}_{i,t-1}}{\mathrm{AT}_{i,t-1}}\)  & \(+1\)\\

\(\mathrm{cinvest}\) &
\(\displaystyle \mathrm{CINVEST}_{i,t}=
\frac{\mathrm{PPENTQ}_{i,t}-\mathrm{PPENTQ}_{i,t-1}}{\mathrm{SALEQ}_{i,t}}
-\frac{1}{3}\sum_{k=1}^{3}
\frac{\mathrm{PPENTQ}_{i,t-k}-\mathrm{PPENTQ}_{i,t-k-1}}{\mathrm{SALEQ}_{i,t-k}}\)  & \(-1\)\\

\(\mathrm{depr}\) &
\(\displaystyle \mathrm{DEPR}_{i,t}=\frac{\mathrm{DP}_{i,t}}{\mathrm{PPENT}_{i,t}}\)  & \(+1\)\\

\(\mathrm{dolvol}\) &
\(\displaystyle \mathrm{DOLVOL}_{i,t}=\log(\mathrm{VOL}_{i,t-2}\mathrm{PRC}_{i,t-2})\)  & \(-1\)\\

\(\mathrm{dy}\) &
\(\displaystyle \mathrm{RETDY}_{i,t}=\mathrm{RET}_{i,t}-\mathrm{RETX}_{i,t},\quad
\mathrm{MDIVPAY}_{i,t}=\mathrm{RETDY}_{i,t}\mathrm{ME}_{i,t-1},\quad
\mathrm{DY}_{i,t}=\frac{\sum_{s=t-11}^{t}\mathrm{MDIVPAY}_{i,s}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{ep}\) &
\(\displaystyle \mathrm{EP}_{i,t}=\frac{\mathrm{IB}_{i,t}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{gma}\) &
\(\displaystyle \mathrm{GMA}_{i,t}=\frac{\mathrm{REVT}_{i,t}-\mathrm{COGS}_{i,t}}{\mathrm{AT}_{i,t-1}}\)  & \(+1\)\\

\(\mathrm{grltnoa}\) &
\(\displaystyle
\begin{aligned}
\mathrm{LTNOA}_{i,t}&=(\mathrm{RECT}_{i,t}+\mathrm{INVT}_{i,t}+\mathrm{PPENT}_{i,t}+\mathrm{ACO}_{i,t}+\mathrm{INTAN}_{i,t}+\mathrm{AO}_{i,t})\\
&\quad-(\mathrm{AP}_{i,t}+\mathrm{LCO}_{i,t}+\mathrm{LO}_{i,t}),\\
\Delta\mathrm{WCNOA}_{i,t}&=(\mathrm{RECT}_{i,t}-\mathrm{RECT}_{i,t-1})+(\mathrm{INVT}_{i,t}-\mathrm{INVT}_{i,t-1})+(\mathrm{ACO}_{i,t}-\mathrm{ACO}_{i,t-1})\\
&\quad-(\mathrm{AP}_{i,t}-\mathrm{AP}_{i,t-1})-(\mathrm{LCO}_{i,t}-\mathrm{LCO}_{i,t-1}),\\
\mathrm{GRLTNOA}_{i,t}&=
\frac{\mathrm{LTNOA}_{i,t}}{\mathrm{AT}_{i,t}}
-\frac{\mathrm{LTNOA}_{i,t-1}}{\mathrm{AT}_{i,t-1}}
-\frac{\Delta\mathrm{WCNOA}_{i,t}}{\frac{1}{2}(\mathrm{AT}_{i,t}+\mathrm{AT}_{i,t-1})}.
\end{aligned}\)  & \(+1\)\\

\(\mathrm{herf}\) &
\(\displaystyle S_{j,t}=\sum_{i\in j}\mathrm{SALE}_{i,t},\quad
\mathrm{HERF}_{j,t}=\sum_{i\in j}\left(\frac{\mathrm{SALE}_{i,t}}{S_{j,t}}\right)^2\)  & \(-1\)\\

\(\mathrm{hire}\) &
\(\displaystyle \mathrm{HIRE}_{i,t}=\frac{\mathrm{EMP}_{i,t}-\mathrm{EMP}_{i,t-1}}{\mathrm{EMP}_{i,t-1}}\)  & \(-1\)\\

\(\mathrm{ill}\) &
\(\displaystyle \mathrm{ILLIQ}_{i,d}=\frac{|\mathrm{RET}_{i,d}|}{|\mathrm{PRC}_{i,d}|\mathrm{VOL}_{i,d}}\)  & \(+1\)\\

\(\mathrm{lev}\) &
\(\displaystyle \mathrm{LEV}_{i,t}=\frac{\mathrm{LT}_{i,t}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{lgr}\) &
\(\displaystyle \mathrm{LGR}_{i,t}=\frac{\mathrm{LT}_{i,t}}{\mathrm{LT}_{i,t-1}}-1\)  & \(-1\)\\

\(\mathrm{maxret}\) &
\(\displaystyle \mathrm{MAXRET}_{i,t}=\max_{d\in D_t}\mathrm{RET}_{i,d}\)  & \(-1\)\\

\(\mathrm{me}\) &
\(\displaystyle \mathrm{ME}_{i,t}=\mathrm{PRC}_{i,t}\mathrm{SHROUT}_{i,t}\).  & \(-1\)\\

\(\mathrm{me\_ia}\) &
\(\displaystyle \mathrm{ME\_IA}_{i,t}=\mathrm{ME}_{i,t}-\overline{\mathrm{ME}}_{\mathrm{industry},t}\)  & \(-1\)\\

\(\mathrm{mom12m}\) &
\(\displaystyle \mathrm{MOM12M}_{i,t}=\prod_{k=1}^{11}(1+\mathrm{RET}_{i,t-k})-1\)  & \(+1\)\\

\(\mathrm{mom1m}\) &
\(\displaystyle \mathrm{MOM1M}_{i,t}=\mathrm{RET}_{i,t}\)  & \(-1\)\\

\(\mathrm{mom36m}\) &
\(\displaystyle \mathrm{MOM36M}_{i,t}=\prod_{k=12}^{35}(1+\mathrm{RET}_{i,t-k})-1\)  & \(-1\)\\

\(\mathrm{mom60m}\) &
\(\displaystyle \mathrm{MOM60M}_{i,t}=\prod_{k=12}^{59}(1+\mathrm{RET}_{i,t-k})-1\)  & \(-1\)\\

\(\mathrm{mom6m}\) &
\(\displaystyle \mathrm{MOM6M}_{i,t}=\prod_{k=1}^{5}(1+\mathrm{RET}_{i,t-k})-1\)  & \(+1\)\\

\(\mathrm{ni}\) &
\(\displaystyle \mathrm{NI}_{i,t}=\log(\mathrm{CSHO}_{i,t}\mathrm{AJEX}_{i,t})
-\log(\mathrm{CSHO}_{i,t-1}\mathrm{AJEX}_{i,t-1})\)  & \(-1\)\\

\(\mathrm{nincr}\) &
\(\displaystyle \mathrm{NINCR}_{i,t}=\sum_{k=1}^{8}\prod_{j=0}^{k-1}\mathbf{1}\{\mathrm{IBQ}_{i,t-j}>\mathrm{IBQ}_{i,t-j-1}\}\)  & \(+1\)\\

\(\mathrm{noa}\) &
\(\displaystyle \mathrm{NOA}_{i,t}=
\frac{(\mathrm{AT}_{i,t}-\mathrm{CHE}_{i,t}-\mathrm{IVAO}_{i,t})
-(\mathrm{AT}_{i,t}-\mathrm{DLC}_{i,t}-\mathrm{DLTT}_{i,t}-\mathrm{MIB}_{i,t}-\mathrm{PSTK}_{i,t}-\mathrm{CEQ}_{i,t})}
{\mathrm{AT}_{i,t-1}}\)  & \(-1\)\\

\(\mathrm{op}\) &
\(\displaystyle \mathrm{OP}_{i,t}=\frac{\mathrm{REVT}_{i,t}-\mathrm{COGS}_{i,t}-\mathrm{XSGA}_{i,t}-\mathrm{XINT}_{i,t}}{\mathrm{BE}_{i,t}}\)  & \(+1\)\\

\(\mathrm{pctacc}\) &
\(\displaystyle \mathrm{PCTACC}_{i,t}=\frac{\mathrm{ACC}_{i,t}}{|\mathrm{NI}_{i,t}|}\)  & \(-1\)\\

\(\mathrm{pm}\) &
\(\displaystyle \mathrm{PM}_{i,t}=\frac{\mathrm{OIADP}_{i,t}}{\mathrm{SALE}_{i,t}}\)  & \(+1\)\\

\(\mathrm{pscore}\) &
\(\displaystyle \mathrm{PSCORE}_{i,t}=\sum_{j=1}^{9}p_{j,i,t}\), where the nine
binary indicators test profitability, operating cash flow, leverage, liquidity,
margin, turnover, and equity issuance conditions.  & \(+1\)\\

\(\mathrm{rd\_sale}\) &
\(\displaystyle \mathrm{RD\_SALE}_{i,t}=\frac{\mathrm{XRD}_{i,t}}{\mathrm{SALE}_{i,t}}\)  & \(+1\)\\

\(\mathrm{rdm}\) &
\(\displaystyle \mathrm{RDM}_{i,t}=\frac{\mathrm{XRD}_{i,t}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{re}\) &
\(\displaystyle \mathrm{RE}_{i,t}=\frac{1}{n}\sum_{k=1}^{n}\mathrm{MONTHLY\_REVISION}_{i,t-k}\),
where \(n=3,4,5,6\) for count \(4,5,6,\ge 7\), respectively.  & \(+1\)\\

\(\mathrm{rna}\) &
\(\displaystyle \mathrm{RNA}_{i,t}=\frac{\mathrm{OIADP}_{i,t}}{\mathrm{NOA}^{raw}_{i,t-1}}\)  & \(+1\)\\

\(\mathrm{roa}\) &
\(\displaystyle \mathrm{ROA}_{i,t}=\frac{\mathrm{IB}_{i,t}}{\mathrm{AT}_{i,t-1}}\).  & \(+1\)\\

\(\mathrm{roe}\) &
\(\displaystyle \mathrm{ROE}_{i,t}=\frac{\mathrm{IB}_{i,t}}{\mathrm{CEQ}_{i,t-1}}\)  & \(+1\)\\

\(\mathrm{rsup}\) &
\(\displaystyle \mathrm{RSUP}_{i,t}=\frac{\mathrm{SALE}_{i,t}-\mathrm{SALE}_{i,t-1}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{rvar\_capm}\) &
\(\displaystyle \mathrm{RVAR\_CAPM}_{i,t}=\frac{1}{w-1}\sum_{s=t-w+1}^{t}(\varepsilon_{i,s}-\overline{\varepsilon}_{i,t})^2\),
where \(\varepsilon\) is from a rolling CAPM excess-return regression.  & \(-1\)\\

\(\mathrm{rvar\_ff3}\) &
\(\displaystyle \mathrm{RVAR\_FF3}_{i,t}=\frac{1}{w-1}\sum_{s=t-w+1}^{t}(\varepsilon_{i,s}-\overline{\varepsilon}_{i,t})^2\),
where \(\varepsilon\) is from a rolling Fama--French three-factor regression.  & \(-1\)\\

\(\mathrm{seas1a}\) &
\(\displaystyle \mathrm{SEAS1A}_{i,t}=\mathrm{RET}_{i,t-11}\)  & \(+1\)\\

\(\mathrm{sgr}\) &
\(\displaystyle \mathrm{SGR}_{i,t}=\frac{\mathrm{SALE}_{i,t}}{\mathrm{SALE}_{i,t-1}}-1\)  & \(+1\)\\

\(\mathrm{sp}\) &
\(\displaystyle \mathrm{SP}_{i,t}=\frac{\mathrm{SALE}_{i,t}}{\mathrm{ME}_{i,t}}\)  & \(+1\)\\

\(\mathrm{std\_dolvol}\) &
\(\displaystyle \mathrm{STD\_DOLVOL}_{i,t}=\operatorname{Std}_{s=t-w+1}^{t}
\left(\log|\mathrm{VOL}_{i,s}\mathrm{PRC}_{i,s}|\right)\)  & \(-1\)\\

\(\mathrm{std\_turn}\) &
\(\displaystyle \mathrm{STD\_TURN}_{i,t}=\operatorname{Std}_{s=t-w+1}^{t}
\left(\frac{\mathrm{VOL}_{i,s}}{\mathrm{SHROUT}_{i,s}}\right)\)  & \(-1\)\\

\(\mathrm{sue}\) &
\(\displaystyle \mathrm{SUE}_{i,t}=\frac{\mathrm{EPS}_{i,t}-\mathrm{EPS}_{i,t-4}}{\mathrm{SUE\_STD}_{i,t}},\quad
\mathrm{EPS}_{i,t}=\frac{\mathrm{EPSPXQ}_{i,t}}{\mathrm{AJEXQ}_{i,t}}\)  & \(+1\)\\

\(\mathrm{svar}\) &
\(\displaystyle \mathrm{RVAR\_MEAN}_{i,t}=\frac{1}{w-1}\sum_{s=t-w+1}^{t}
(\mathrm{RET}_{i,s}-\overline{\mathrm{RET}}_{i,t})^2\)  & \(-1\)\\

\(\mathrm{turn}\) &
\(\displaystyle \mathrm{TURN}_{i,t}=\frac{\mathrm{VOL}_{i,t}}{\mathrm{SHROUT}_{i,t}}\).  & \(-1\)\\

\(\mathrm{zerotrade}\) &
\(\displaystyle \mathrm{ZEROTRADE}_{i,t}=
\left(\sum_{s=t-w+1}^{t}\mathrm{COUNTZERO}_{i,s}+
\frac{1}{\sum_{s=t-w+1}^{t}\mathrm{TURN}_{i,s}}/11000\right)
\frac{21\cdot 3}{w}\)  & \(+1\)\\
